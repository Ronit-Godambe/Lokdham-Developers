        /* =====================================
           UPLOAD IMAGE TO SUPABASE STORAGE
        ===================================== */

        async function uploadBlogImage(file) {

            if (!file) {

                return null;

            }


            const fileExtension =
                file.name.split(".").pop();


            const fileName =
                `blogs/${Date.now()}-${Math.random()
                    .toString(36)
                    .substring(2)}.${fileExtension}`;


            const {
                error
            } = await supabase
                .storage
                .from("blog-images")
                .upload(
                    fileName,
                    file
                );


            if (error) {

                console.error(
                    "Image upload error:",
                    error
                );

                throw error;

            }


            const {
                data
            } = supabase
                .storage
                .from("blog-images")
                .getPublicUrl(
                    fileName
                );


            return data.publicUrl;

        }

        /* =====================================
   SAVE BLOG
===================================== */

        saveBlogBtn.addEventListener(
            "click",
            async function () {

                try {

                    /* =====================================
                       BASIC VALIDATION
                    ===================================== */

                    if (
                        !blogTitle.value.trim()
                    ) {

                        alert(
                            "Please enter a blog title."
                        );

                        return;

                    }


                    if (
                        !blogSlug.value.trim()
                    ) {

                        alert(
                            "Please enter a blog URL slug."
                        );

                        return;

                    }


                    /* =====================================
                       SAVE BUTTON STATE
                    ===================================== */

                    const originalButtonText =
                        saveBlogBtn.textContent;


                    saveBlogBtn.disabled =
                        true;


                    saveBlogBtn.textContent =
                        "Saving...";


                    /* =====================================
                       COVER IMAGE
                    ===================================== */

                    let coverImageUrl =
                        null;


                    const coverFile =
                        coverImageInput.files[0];


                    if (coverFile) {

                        coverImageUrl =
                            await uploadBlogImage(
                                coverFile
                            );

                    }


                    /* =====================================
                       HERO IMAGE
                    ===================================== */

                    let heroImageUrl =
                        null;


                    const heroFile =
                        heroImageInput.files[0];


                    if (heroFile) {

                        heroImageUrl =
                            await uploadBlogImage(
                                heroFile
                            );

                    }


                    /* =====================================
                       COLLECT ARTICLE BLOCKS
                    ===================================== */

                    const contentBlocks =
                        [];


                    const blocks =
                        blogContentBlocks.querySelectorAll(
                            ".content-block"
                        );


                    for (
                        const block
                        of blocks
                    ) {

                        /* HEADING */

                        if (
                            block.classList.contains(
                                "heading-block"
                            )
                        ) {

                            const input =
                                block.querySelector(
                                    ".content-block-heading-input"
                                );


                            contentBlocks.push({

                                type:
                                    "heading",

                                content:
                                    input.value.trim()

                            });

                        }


                        /* PARAGRAPH */

                        if (
                            block.classList.contains(
                                "paragraph-block"
                            )
                        ) {

                            const textarea =
                                block.querySelector(
                                    ".content-block-paragraph-input"
                                );


                            contentBlocks.push({

                                type:
                                    "paragraph",

                                content:
                                    textarea.value.trim()

                            });

                        }


                        /* IMAGE */

                        if (
                            block.classList.contains(
                                "image-block"
                            )
                        ) {

                            const imageInput =
                                block.querySelector(
                                    ".content-block-image-input"
                                );


                            const imageFile =
                                imageInput.files[0];


                            let imageUrl =
                                null;


                            if (imageFile) {

                                imageUrl =
                                    await uploadBlogImage(
                                        imageFile
                                    );

                            }


                            if (imageUrl) {

                                contentBlocks.push({

                                    type:
                                        "image",

                                    content:
                                        imageUrl

                                });

                            }

                        }

                    }


                    /* =====================================
                       CREATE BLOG DATA
                    ===================================== */

                    const blogData = {

                        title:
                            blogTitle.value.trim(),

                        category:
                            blogCategory.value.trim(),

                        description:
                            blogDescription.value.trim(),

                        slug:
                            blogSlug.value
                                .trim()
                                .toLowerCase(),

                        status:
                            blogStatus.value,

                        cover_image:
                            coverImageUrl,

                        hero_image:
                            heroImageUrl,

                        content:
                            contentBlocks,

                        seo_title:
                            blogSeoTitle.value.trim(),

                        seo_description:
                            blogSeoDescription
                                .value
                                .trim()

                    };


                    /* =====================================
                    SAVE OR UPDATE BLOG
                    ===================================== */

                    let saveError;


                    if (
                        currentEditingBlogId
                    ) {

                        /* UPDATE EXISTING BLOG */

                        const {
                            error
                        } = await supabase
                            .from(
                                "blogs"
                            )
                            .update(
                                blogData
                            )
                            .eq(
                                "id",
                                currentEditingBlogId
                            );


                        saveError =
                            error;

                    } else {

                        /* CREATE NEW BLOG */

                        const {
                            error
                        } = await supabase
                            .from(
                                "blogs"
                            )
                            .insert(
                                blogData
                            );


                        saveError =
                            error;

                    }


                    if (
                        saveError
                    ) {

                        throw saveError;

                    }


                    if (
                        currentEditingBlogId
                    ) {

                        alert(
                            "Blog updated successfully!"
                        );

                    } else {

                        alert(
                            "Blog saved successfully!"
                        );

                    }


                    currentEditingBlogId =
                        null;


                    showBlogs();


                    showBlogs();

                }

                catch (
                error
                ) {

                    console.error(
                        "Error saving blog:",
                        error
                    );


                    alert(
                        "Something went wrong while saving the blog. Check the console."
                    );

                }

                finally {

                    saveBlogBtn.disabled =
                        false;


                    saveBlogBtn.textContent =
                        "Save Blog";

                }

            }
        );

        /* =====================================
EDIT BLOG
===================================== */

        async function editBlog(blogId) {

            try {

                const {
                    data: blog,
                    error
                } = await supabase
                    .from("blogs")
                    .select("*")
                    .eq(
                        "id",
                        blogId
                    )
                    .single();


                if (error) {

                    throw error;

                }


                /* STORE EDITING BLOG ID */

                currentEditingBlogId =
                    blog.id;


                /* OPEN EDITOR */

                showBlogEditor();


                blogEditorTitle.textContent =
                    "Edit Blog";


                /* BASIC DETAILS */

                blogTitle.value =
                    blog.title || "";

                blogCategory.value =
                    blog.category || "";

                blogDescription.value =
                    blog.description || "";

                blogSlug.value =
                    blog.slug || "";

                blogStatus.value =
                    blog.status || "";

                blogSeoTitle.value =
                    blog.seo_title || "";

                blogSeoDescription.value =
                    blog.seo_description || "";


                /* COVER IMAGE */

                if (blog.cover_image) {

                    coverImagePreview.innerHTML =
                        `<img src="${blog.cover_image}" alt="">`;

                } else {

                    coverImagePreview.innerHTML =
                        "No image selected";

                }


                /* HERO IMAGE */

                if (blog.hero_image) {

                    heroImagePreview.innerHTML =
                        `<img src="${blog.hero_image}" alt="">`;

                } else {

                    heroImagePreview.innerHTML =
                        "No image selected";

                }


                /* CLEAR OLD CONTENT */

                blogContentBlocks.innerHTML =
                    "";


                /* LOAD CONTENT BLOCKS */

                if (
                    Array.isArray(blog.content)
                ) {

                    blog.content.forEach(
                        function (block) {

                            if (
                                block.type === "heading"
                            ) {

                                addHeadingBlock(
                                    block.content
                                );

                            }


                            if (
                                block.type === "paragraph"
                            ) {

                                addParagraphBlock(
                                    block.content
                                );

                            }


                            if (
                                block.type === "image"
                            ) {

                                addImageBlock(
                                    block.content
                                );

                            }

                        }
                    );

                }


                updateContentEmptyState();

            }

            catch (error) {

                console.error(
                    "Error loading blog:",
                    error
                );

                alert(
                    "Unable to load this blog."
                );

            }

        }

                    .order(
                        "plot_number",
                        {
                            ascending: true
                        }
                    );


            loading.style.display =
                "none";


            if (error) {

                console.error(
                    error
                );


                plotGrid.innerHTML =
                    "<p>Error loading plots.</p>";


                return;

            }


            if (
                !data ||
                data.length === 0
            ) {

                plotGrid.innerHTML =
                    "<p>No plots found.</p>";


                return;

            }


            data.forEach(
                function (plot) {


                    originalStatuses[
                        plot.id
                    ]
                        =
                        plot.status;


                    const card =
                        document.createElement(
                            "div"
                        );


                    card.className =
                        "plot-card";


                    card.innerHTML =
                        `

                        <div class="plot-card-top">

                            <div class="plot-number">

                                ${plot.plot_number}

                            </div>

                        </div>


                        <div class="plot-details">

                            <h3>

                                ${plot.plot_size}

                            </h3>


                            <p>

                                Road:
                                ${plot.road}

                            </p>

                        </div>

                            id="status-${plot.id}"
                            data-id="${plot.id}"
                        >

                            <option
                                value="available"
                                ${plot.status === "available"
                            ? "selected"
                            : ""
                        }
                            >

                                Available

                            </option>


                            <option
                                value="booked"
                                ${plot.status === "booked"
                            ? "selected"
                            : ""
                        }
                            >

                                Booked

                            </option>


                            <option
                                value="soldout"
                                ${plot.status === "soldout"
                            ? "selected"
                            : ""
                        }
                            >

                                Sold Out

                    );

        /* =====================================
           SAVE ALL CHANGES
        ====================================== */

        saveAllBtn.addEventListener(
            "click",
            async function () {


                const selects =
                    document.querySelectorAll(
                        "#plotGrid select"
                    );


                const changes =
                    [];


                selects.forEach(
                    function (select) {


                        const id =
                            select.dataset.id;


                        const newStatus =
                            select.value;


                        if (
                            originalStatuses[id]
                            !==
                            newStatus
                        ) {

                            changes.push({
                                id: id,
                                status: newStatus
                            });

                        }

                    }
                );


                if (
                    changes.length === 0
                ) {

                    showMessage(
                        "No changes to save."
                    );


                    return;

                }


                saveAllBtn.disabled =
                    true;


                saveAllBtn.textContent =
                    "Saving...";


                let hasError =
                    false;


                for (
                    const change
                    of
                    changes
                ) {


                    const {
                        error
                    }
                        =
                        await supabase

                            .from(
                                "kelad_greens_plots"
                            )

                            .update({
                                status:
                                    change.status
                            })

                            .eq(
                                "id",
                                change.id
                            );


                    if (
                        error
                    ) {

                        console.error(
                            error
                        );


                        hasError =
                            true;

                    }

                }


                saveAllBtn.disabled =
                    false;


                saveAllBtn.textContent =
                    "Save All Changes";


                if (
                    hasError
                ) {

                    showMessage(
                        "Some changes could not be saved."
                    );


                    return;

                }


                changes.forEach(
                    function (change) {

                        originalStatuses[
                            change.id
                        ]
                            =
                            change.status;

                    }
                );


                showMessage(
                    "All changes saved successfully!"
                );

            }
        );

        loginForm.addEventListener(
            "submit",
            async function (event) {

                event.preventDefault();


                const loginValue =
                    document.getElementById(
                        "loginInput"
                    ).value
                        .trim()
                        .toLowerCase();


                const password =
                    document.getElementById(
                        "password"
                    ).value;


                /* =====================================
                   USERNAME → EMAIL MAP
                ====================================== */

                const usernameMap = {

                    lokdham: "lokdhamdevelopers@gmail.com",
                    rgdev: "ronitgodambe@gmail.com"

                };


                let email;


                /* If user entered an email */
                if (
                    loginValue.includes("@")
                ) {

                    email = loginValue;

                }

                /* If user entered a username */
                else {

                    email =
                        usernameMap[loginValue];

                }


                /* Invalid username */

                if (!email) {

                    loginMessage.textContent =
                        "Invalid username or email.";

                    return;

                }


                loginMessage.textContent =
                    "Logging in...";


                const {
                    error
                }
                    =
                    await supabase
                        .auth
                        .signInWithPassword({

                            email: email,

                            password: password

                        });


                if (
                    error
                ) {

                    console.error(
                        error
                    );


                    loginMessage.textContent =
                        "Invalid username, email or password.";

                    return;

                }


                loginMessage.textContent =
                    "";


                loginScreen.style.display =
                    "none";


                appScreen.style.display =
                    "block";


                showAdminHome();

            }
        );



        /* =====================================
           CHECK EXISTING SESSION
        ====================================== */

        async function checkLogin() {


            const {
                data:
                {
                    session
                }
            }
                =
                await supabase

                    .auth

                    .getSession();

            ) {

                    "none";


                appScreen.style.display =
                    "block";


                showAdminHome();

            }

            else {

                loginScreen.style.display =
                    "flex";


                appScreen.style.display =
                    "none";

            }

        }
